<script>

let course;
let currentStep = 0;

function init() {

  google.script.run
    .withSuccessHandler(loadCourse)
    .getCourse();

}

function loadCourse(data) {

  course = data;

  document.getElementById('courseTitle').textContent =
    data.title;

  document.getElementById('courseDescription').textContent =
    data.description;

  renderSteps();

  loadStep(0);

}

function renderSteps() {

  const container =
    document.getElementById('steps');

  container.innerHTML = '';

  course.steps.forEach((step, index) => {

    const element =
      document.createElement('div');

    element.className = 'step';

    element.textContent =
      (index + 1) + '. ' + step.title;

    element.onclick = () =>
      loadStep(index);

    container.appendChild(element);

  });

}

function loadStep(index) {

  currentStep = index;

  document
    .querySelectorAll('.step')
    .forEach((element, i) => {

      element.classList.toggle(
        'active',
        i === index
      );

    });

  const step =
    course.steps[index];

  google.script.run
    .withSuccessHandler(renderStep)
    .getStep(step.id);

}

function renderStep(step) {

  document.getElementById('editor').value =
    step.initialCode || '';

  document.getElementById('fileName').textContent =
    step.file || 'workspace';

}

function verify() {

  writeTerminal(
    'Running verification...\n'
  );

  google.script.run
    .withSuccessHandler(result => {

      writeTerminal(
        result.message + '\n'
      );

    })
    .verifyStep(
      course.steps[currentStep].id,
      document.getElementById('editor').value
    );

}

function nextStep() {

  if (currentStep < course.steps.length - 1) {

    loadStep(currentStep + 1);

  }

}

function terminalKey(event) {

  if (event.key !== 'Enter') {
    return;
  }

  const input =
    document.getElementById('terminalInput');

  const command =
    input.value.trim();

  if (!command) {
    return;
  }

  writeTerminal('$ ' + command + '\n');

  input.value = '';

  google.script.run
    .withSuccessHandler(result => {

      writeTerminal(result + '\n');

    })
    .executeCommand(command);

}

function writeTerminal(text) {

  const output =
    document.getElementById('terminalOutput');

  output.textContent += text;

  output.scrollTop =
    output.scrollHeight;

}

init();
